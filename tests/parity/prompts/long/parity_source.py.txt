# ===== tests/parity/logits.py =====
"""Compact per-prompt logit summaries, and the metrics that compare two of them.

Full logits are too big to keep for every position (vocab 248,320 x 4 bytes ~ 1 MB per
position), so a ``LogitSummary`` keeps, for each position, the argmax, the top-k logits and the
log-sum-exp, and full logits only at a few sampled positions (the last ones, where decoding
starts, plus random ones). KL and max |diff| are computed on those full positions; argmax
agreement on every position.

Logits are produced by applying the LM head to the final hidden states in chunks of positions,
so a 32k-token prompt never materialises a [32k, vocab] tensor. Reference and candidate must go
through the same ``summarize`` for the comparison to be fair.
"""

from __future__ import annotations

from collections.abc import Iterable
from dataclasses import dataclass, fields
from pathlib import Path

import torch
from safetensors.torch import load_file, save_file

TOPK = 20


@dataclass
class LogitSummary:
    argmax: torch.Tensor  # [seq] int64
    topk_ids: torch.Tensor  # [seq, k] int64
    topk_vals: torch.Tensor  # [seq, k] float32 (raw logits)
    lse: torch.Tensor  # [seq] float32
    full_pos: torch.Tensor  # [n] int64
    full: torch.Tensor  # [n, vocab] float32

    def save(self, path: Path) -> None:
        path.parent.mkdir(parents=True, exist_ok=True)
        save_file({f.name: getattr(self, f.name).contiguous() for f in fields(self)}, str(path))

    @classmethod
    def load(cls, path: Path) -> LogitSummary:
        return cls(**load_file(str(path)))


def full_positions(seq_len: int, n_last: int = 16, n_random: int = 16, seed: int = 0) -> list[int]:
    """Sorted positions whose full logits are kept: the last ``n_last`` plus random others."""
    last = set(range(max(0, seq_len - n_last), seq_len))
    rest = [p for p in range(seq_len) if p not in last]
    g = torch.Generator().manual_seed(seed)
    pick = torch.randperm(len(rest), generator=g)[:n_random].tolist()
    return sorted(last | {rest[i] for i in pick})


def summarize(chunks: Iterable[torch.Tensor], full_pos: list[int], k: int = TOPK) -> LogitSummary:
    """Summarise logits given as consecutive ``[chunk, vocab]`` blocks covering every position."""
    want = set(full_pos)
    argmax, tid, tval, lse, full = [], [], [], [], []
    start = 0
    for c in chunks:
        c = c.float()
        argmax.append(c.argmax(-1).cpu())
        v, i = c.topk(k, dim=-1)
        tid.append(i.cpu())
        tval.append(v.cpu())
        lse.append(torch.logsumexp(c, -1).cpu())
        for j in range(c.shape[0]):
            if start + j in want:
                full.append(c[j].cpu())
        start += c.shape[0]
    return LogitSummary(
        argmax=torch.cat(argmax),
        topk_ids=torch.cat(tid),
        topk_vals=torch.cat(tval),
        lse=torch.cat(lse),
        full_pos=torch.tensor(sorted(want), dtype=torch.int64),
        full=torch.stack(full) if full else torch.empty(0),
    )


def compare(ref: LogitSummary, cand: LogitSummary) -> dict[str, float]:
    """Metrics of ``cand`` against ``ref`` for one prompt."""
    if ref.argmax.shape != cand.argmax.shape:
        raise ValueError(f"length mismatch {ref.argmax.shape} vs {cand.argmax.shape}")
    if not torch.equal(ref.full_pos, cand.full_pos):
        raise ValueError("full_pos differ: summaries built with different position samples")
    diff = (ref.full - cand.full).abs()
    lp_ref = torch.log_softmax(ref.full, -1)
    lp_cand = torch.log_softmax(cand.full, -1)
    kl = (lp_ref.exp() * (lp_ref - lp_cand)).sum(-1)  # KL(ref || cand) per position
    mism = ref.argmax != cand.argmax
    return {
        "positions": int(ref.argmax.numel()),
        "argmax_mismatch": int(mism.sum()),
        "argmax_mismatch_rate": float(mism.float().mean()),
        "first_mismatch_pos": int(mism.nonzero()[0, 0]) if mism.any() else -1,
        "max_abs_logit": float(diff.max()),
        "mean_abs_logit": float(diff.mean()),
        "kl_mean": float(kl.mean()),
        "kl_max": float(kl.max()),
        "lse_max_abs": float((ref.lse - cand.lse).abs().max()),
        "bitwise_equal": bool(torch.equal(ref.full, cand.full) and not mism.any()),
    }


def aggregate(per_prompt: dict[str, dict[str, float]]) -> dict[str, float]:
    """Worst case over prompts (max) for diffs, totals for counts."""
    vals = list(per_prompt.values())
    pos = sum(v["positions"] for v in vals)
    mis = sum(v["argmax_mismatch"] for v in vals)
    return {
        "prompts": len(vals),
        "positions": pos,
        "argmax_mismatch": mis,
        "argmax_mismatch_rate": mis / pos if pos else 0.0,
        "max_abs_logit": max(v["max_abs_logit"] for v in vals),
        "mean_abs_logit": sum(v["mean_abs_logit"] for v in vals) / len(vals),
        "kl_mean": sum(v["kl_mean"] for v in vals) / len(vals),
        "kl_max": max(v["kl_max"] for v in vals),
        "bitwise_equal": all(v["bitwise_equal"] for v in vals),
    }
# ===== tests/parity/candidates.py =====
"""Things whose logits can be compared: HF Transformers today, the vinfer engine later.

A candidate turns token ids into ``LogitSummary`` objects (``tests/parity/logits.py``).
HF knobs exposed here are exactly the noise-floor dimensions (``noise_floor.py``): dtype,
batch size, padding side, request order, attention implementation.
"""

from __future__ import annotations

from collections.abc import Iterator, Sequence
from dataclasses import dataclass
from typing import Literal, Protocol

import torch

from tests.parity.logits import LogitSummary, full_positions, summarize

DTYPES = {"float32": torch.float32, "bfloat16": torch.bfloat16, "float16": torch.float16}


class Candidate(Protocol):
    name: str

    def summaries(self, prompts: Sequence[list[int]]) -> list[LogitSummary]: ...


@dataclass
class HFConfig:
    model: str
    revision: str | None = None
    dtype: str = "bfloat16"
    device: str = "cuda"
    attn_implementation: str = "sdpa"
    batch_size: int = 1
    padding_side: Literal["left", "right"] = "right"
    order_seed: int | None = None  # None = prompt order; int = shuffled order
    chunk: int = 1024  # positions per LM-head chunk

    @property
    def name(self) -> str:
        return (
            f"hf[{self.dtype},{self.attn_implementation},bs={self.batch_size},"
            f"pad={self.padding_side},order={self.order_seed}]"
        )


def environment() -> dict[str, str]:
    """Everything that can change HF logits bit-for-bit besides the model and dtype."""
    import transformers

    return {
        "torch": torch.__version__,
        "cuda": str(torch.version.cuda),
        "transformers": transformers.__version__,
        "gpu": torch.cuda.get_device_name(0) if torch.cuda.is_available() else "cpu",
        "gdn_backend": gdn_backend(),
    }


def gdn_backend() -> str:
    """Which Gated DeltaNet implementation HF will use: the fla kernels if importable."""
    try:
        import fla  # noqa: F401

        return "fla"
    except ImportError:
        return "torch"


class HFCandidate:
    def __init__(self, cfg: HFConfig, model: torch.nn.Module | None = None) -> None:
        self.cfg = cfg
        self.name = cfg.name
        if model is None:
            from transformers import AutoModelForCausalLM

            model = AutoModelForCausalLM.from_pretrained(
                cfg.model,
                revision=cfg.revision,
                dtype=DTYPES[cfg.dtype],
                attn_implementation=cfg.attn_implementation,
                device_map=cfg.device,  # load straight to the device, no CPU copy first
            )
        self.model = model.eval()

    def _chunks(self, hidden: torch.Tensor) -> Iterator[torch.Tensor]:
        for s in range(0, hidden.shape[0], self.cfg.chunk):
            yield self.model.lm_head(hidden[s : s + self.cfg.chunk])

    @torch.inference_mode()
    def summaries(self, prompts: Sequence[list[int]]) -> list[LogitSummary]:
        order = list(range(len(prompts)))
        if self.cfg.order_seed is not None:
            g = torch.Generator().manual_seed(self.cfg.order_seed)
            order = torch.randperm(len(prompts), generator=g).tolist()
        out: dict[int, LogitSummary] = {}
        bs = self.cfg.batch_size
        dev = next(self.model.parameters()).device
        for b in range(0, len(order), bs):
            idx = order[b : b + bs]
            seqs = [prompts[i] for i in idx]
            width = max(len(s) for s in seqs)
            ids = torch.zeros(len(seqs), width, dtype=torch.long)
            mask = torch.zeros(len(seqs), width, dtype=torch.long)
            for r, s in enumerate(seqs):
                if self.cfg.padding_side == "right":
                    ids[r, : len(s)] = torch.tensor(s)
                    mask[r, : len(s)] = 1
                else:
                    ids[r, width - len(s) :] = torch.tensor(s)
                    mask[r, width - len(s) :] = 1
            kwargs = {"input_ids": ids.to(dev)}
            if len(seqs) > 1:
                kwargs["attention_mask"] = mask.to(dev)
                # Positions count real tokens only, as HF generate does for left padding.
                kwargs["position_ids"] = (mask.cumsum(-1) - 1).clamp(min=0).to(dev)
            # use_cache=False: no KV / recurrent-state cache is needed for one forward pass.
            hidden = self.model.model(**kwargs, use_cache=False).last_hidden_state
            for r, (i, s) in enumerate(zip(idx, seqs, strict=True)):
                h = (
                    hidden[r, : len(s)]
                    if self.cfg.padding_side == "right"
                    else hidden[r, -len(s) :]
                )
                out[i] = summarize(self._chunks(h), full_positions(len(s)))
        return [out[i] for i in range(len(prompts))]
# ===== tests/parity/noise_floor.py =====
"""Measure the natural logit noise of HF Transformers, to set parity thresholds from.

Compares HF variants against an HF reference on the parity prompt set. Only things that should
not change the result are varied, so every difference is noise the engine will also face:

- batch size {1, 4, 16}, request order (shuffled), padding side (right/left),
- attention implementation (sdpa / eager),
- dtype: bf16 and fp16 runs against an fp32 reference (the "true" logits),
- GDN backend (fla kernels vs torch): not switchable in-process; run this script twice, once
  in an env with ``flash-linear-attention`` installed and once without, and compare the outputs.

Usage (GPU)::

    python -m tests.parity.noise_floor --model Qwen/Qwen3.5-2B --revision <sha>

Output (default ``bench/results/raw/noise-floor-<fla|torch>.json``): one JSON with, for each
variant, the aggregate and per-prompt metrics of ``tests/parity/logits.py:compare``, plus the
environment.
"""

from __future__ import annotations

import argparse
import json
from pathlib import Path

import torch

from bench.harness.config import pinned_revision
from tests.parity import prompt_set
from tests.parity.candidates import HFCandidate, HFConfig, environment, gdn_backend
from tests.parity.logits import LogitSummary, aggregate, compare


def variants(model: str, revision: str | None, dtype: str) -> dict[str, HFConfig]:
    """Configs that should give the same logits as the plain ``dtype`` baseline."""
    base = {"model": model, "revision": revision, "dtype": dtype}
    return {
        "repeat": HFConfig(**base),  # the baseline again: expected bit-identical
        "batch4": HFConfig(**base, batch_size=4),
        "batch16": HFConfig(**base, batch_size=16),
        "batch4-left-pad": HFConfig(**base, batch_size=4, padding_side="left"),
        "batch4-shuffled": HFConfig(**base, batch_size=4, order_seed=1),
        "eager-attention": HFConfig(**base, attn_implementation="eager"),
    }


def summaries(cfg: HFConfig, seqs: list[list[int]]) -> list[LogitSummary]:
    """Load the model for ``cfg``, summarise every prompt, free the GPU."""
    cand = HFCandidate(cfg)
    try:
        return cand.summaries(seqs)
    finally:
        del cand
        torch.cuda.empty_cache()


def metrics(keys: list[str], ref: list[LogitSummary], got: list[LogitSummary]) -> dict:
    per = {k: compare(r, g) for k, r, g in zip(keys, ref, got, strict=True)}
    return {"aggregate": aggregate(per), "per_prompt": per}


def run(model: str, revision: str | None, out_path: Path, dtypes: list[str]) -> dict:
    """Measure every variant; the JSON at ``out_path`` is rewritten after each one."""
    from transformers import AutoTokenizer

    tok = AutoTokenizer.from_pretrained(model, revision=revision)
    ids = prompt_set.tokenize(prompt_set.load("v1"), tok)
    keys, seqs = list(ids), list(ids.values())
    result: dict = {
        "model": model,
        "revision": revision,
        "environment": environment(),
        "prompt_tokens": {k: len(v) for k, v in ids.items()},
        "variants": {},
    }
    out_path.parent.mkdir(parents=True, exist_ok=True)

    def record(label: str, ref: list[LogitSummary], got: list[LogitSummary]) -> None:
        result["variants"][label] = metrics(keys, ref, got)
        print(label, json.dumps(result["variants"][label]["aggregate"]), flush=True)
        out_path.write_text(json.dumps(result, indent=2))

    fp32 = summaries(HFConfig(model, revision, "float32"), seqs)  # closest to exact logits
    for dtype in dtypes:
        base = summaries(HFConfig(model, revision, dtype), seqs)
        record(f"{dtype}-vs-fp32", fp32, base)  # precision gap
        for name, cfg in variants(model, revision, dtype).items():
            record(f"{dtype}:{name}", base, summaries(cfg, seqs))  # same-dtype noise
    return result


def main() -> None:
    p = argparse.ArgumentParser(
        description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter
    )
    p.add_argument("--model", required=True)
    p.add_argument("--revision", help="default: the pin in bench/models.lock")
    p.add_argument("--dtypes", nargs="+", default=["bfloat16", "float16"])
    # Default under bench/results/raw/ (git-ignored): writing into experiments/ mid-session
    # would make the tree dirty and block official benchmark runs until a commit. The GDN
    # backend is in the name, so the runs with and without flash-linear-attention both stay.
    p.add_argument("--out", type=Path, help="default: bench/results/raw/noise-floor-<gdn>.json")
    a = p.parse_args()
    # Same weights as parity and the baselines: never silently fall back to `main`.
    revision = a.revision or pinned_revision(a.model)
    if revision is None:
        raise SystemExit(f"no revision for {a.model}: pin it in bench/models.lock or pass one")
    out = a.out or Path(f"bench/results/raw/noise-floor-{gdn_backend()}.json")
    run(a.model, revision, out, a.dtypes)


if __name__ == "__main__":
    main()
# ===== tests/parity/prompt_set.py =====
"""Load and tokenize the fixed parity prompt set (``prompts/v1.jsonl``).

Kinds: ``short``, ``multilingual``, ``code``, ``chat`` (system + multi-turn), ``math`` go through
the model's chat template; ``raw`` is tokenized as plain text; ``long`` concatenates snapshot
files in ``prompts/long/`` (frozen copies, so the set never changes) and appends an instruction.
Never edit v1: add v2.
"""

from __future__ import annotations

import json
from pathlib import Path
from typing import Any

HERE = Path(__file__).resolve().parent / "prompts"


def load(version: str = "v1") -> list[dict[str, Any]]:
    return [json.loads(x) for x in (HERE / f"{version}.jsonl").read_text().splitlines() if x]


def tokenize(prompts: list[dict[str, Any]], tokenizer: Any) -> dict[str, list[int]]:
    out = {}
    for p in prompts:
        if p["kind"] == "raw":
            ids = tokenizer.encode(p["text"], add_special_tokens=False)
        else:
            if p["kind"] == "long":
                doc = "\n\n".join((HERE / "long" / f).read_text() for f in p["files"])
                messages = [{"role": "user", "content": f"{doc}\n\n{p['instruction']}"}]
            else:
                messages = p["messages"]
            ids = tokenizer.apply_chat_template(messages, add_generation_prompt=True, tokenize=True)
            if hasattr(ids, "keys"):
                ids = ids["input_ids"]
        out[p["id"]] = list(ids)
    return out
